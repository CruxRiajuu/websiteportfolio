<script>
const rotatorImages = [
"https://i.imgur.com/K59fz6W.jpg",
"https://i.imgur.com/PSvIweG.jpg",
"https://i.imgur.com/YyfAQwZ.jpg",
"https://i.imgur.com/83pW10Q.jpeg",
"https://i.imgur.com/1SaK8Ye.png"
];

let rotatorIndex = 0;
let images = [];
let touchStartX = 0;
let touchEndX = 0;
let isFirstBuild = true;

function getDynamicPositions() { const rotator = document.getElementById("photo-rotator"); if (!rotator) return []; const rotatorWidth = rotator.offsetWidth; const isMobile = window.innerWidth <= 768; const xOffset = isMobile ? rotatorWidth * 0.35 : rotatorWidth * 0.35; const zOffsetCenter = isMobile ? 100 : 200; const zOffsetSide = isMobile ? -25 : -50; const scaleSide = isMobile ? 0.65 : 0.7; const rotation = isMobile ? -40 : -45; return [ { transform: `translate(-50%, -50%) translateZ(${zOffsetCenter}px) scale(1)`, opacity: 1, zIndex: 3, cursor: "pointer" }, { transform: `translate(-50%, -50%) rotateY(${rotation}deg) translateX(${xOffset}px) translateZ(${zOffsetSide}px) scale(${scaleSide})`, opacity: 0.4, zIndex: 2, cursor: "default" }, { transform: `translate(-50%, -50%) rotateY(${-rotation}deg) translateX(-${xOffset}px) translateZ(${zOffsetSide}px) scale(${scaleSide})`, opacity: 0.4, zIndex: 2, cursor: "default" } ]; }
function buildRotator() { const container = document.getElementById("photo-rotator"); if (!container) return; if (images.length === 0) { rotatorImages.forEach((src, i) => { const img = document.createElement("img"); img.src = src; img.className = "photo-display"; img.alt = `Gallery image ${i + 1}`; img.onclick = () => { if (i === rotatorIndex) openLightbox(); }; container.appendChild(img); images.push(img); }); } if (isFirstBuild) { images.forEach(img => img.style.transition = 'none'); } const positions = getDynamicPositions(); images.forEach((img, i) => { let posIndex = (i - rotatorIndex + rotatorImages.length) % rotatorImages.length; if (posIndex === 0) { Object.assign(img.style, positions[0]); img.classList.add("center-focus"); } else if (posIndex === 1) { Object.assign(img.style, positions[1]); img.classList.remove("center-focus"); } else if (posIndex === rotatorImages.length - 1) { Object.assign(img.style, positions[2]); img.classList.remove("center-focus"); } else { img.style.opacity = 0; img.style.zIndex = 0; img.style.cursor = "default"; img.style.transform = "translate(-50%, -50%) scale(0)"; img.classList.remove("center-focus"); } }); if (isFirstBuild) { void container.offsetWidth; images.forEach(img => img.style.transition = ''); isFirstBuild = false; } container.style.opacity = 1; }
function rotateNext() { rotatorIndex = (rotatorIndex + 1) % rotatorImages.length; buildRotator(); }
function rotatePrev() { rotatorIndex = (rotatorIndex - 1 + rotatorImages.length) % rotatorImages.length; buildRotator(); }
function openLightbox() { const overlay = document.getElementById("lightbox-overlay"); overlay.style.display = "block"; document.body.style.overflow = "hidden"; populateLightboxContent(); updateLightboxPosition(false); overlay.addEventListener('touchstart', handleTouchStart, { passive: true }); overlay.addEventListener('touchend', handleTouchEnd, { passive: true }); }
function closeLightbox() { const overlay = document.getElementById("lightbox-overlay"); overlay.style.display = "none"; document.body.style.overflow = "auto"; overlay.removeEventListener('touchstart', handleTouchStart); overlay.removeEventListener('touchend', handleTouchEnd); }
function lightboxNext() { rotatorIndex = (rotatorIndex + 1) % rotatorImages.length; updateLightboxPosition(true); buildRotator(); }
function lightboxPrev() { rotatorIndex = (rotatorIndex - 1 + rotatorImages.length) % rotatorImages.length; updateLightboxPosition(true); buildRotator(); }
function updateLightboxPosition(withAnimation = true) { const filmstrip = document.getElementById('lightbox-filmstrip'); const newTransform = `translateX(-${rotatorIndex * 100}%)`; filmstrip.style.transition = withAnimation ? '' : 'none'; filmstrip.style.transform = newTransform; const thumbnails = document.querySelectorAll(".lightbox-thumb"); thumbnails.forEach((thumb, i) => { if (i === rotatorIndex) { thumb.classList.add("active"); thumb.scrollIntoView({ behavior: 'smooth', inline: 'center' }); } else { thumb.classList.remove("active"); } }); }
function populateLightboxContent() { const filmstrip = document.getElementById("lightbox-filmstrip"); const thumbContainer = document.getElementById("lightbox-thumbnails-container"); filmstrip.innerHTML = ""; thumbContainer.innerHTML = ""; rotatorImages.forEach((src, i) => { const wrapper = document.createElement("div"); wrapper.className = "lightbox-image-wrapper"; const img = document.createElement("img"); img.src = src; img.alt = `Lightbox image ${i + 1}`; wrapper.appendChild(img); filmstrip.appendChild(wrapper); const thumb = document.createElement("img"); thumb.src = src; thumb.className = "lightbox-thumb"; thumb.alt = `Thumbnail ${i + 1}`; thumb.onclick = () => { rotatorIndex = i; updateLightboxPosition(true); buildRotator(); }; thumbContainer.appendChild(thumb); }); }
function handleTouchStart(e) { touchStartX = e.changedTouches[0].screenX; }
function handleTouchEnd(e) { touchEndX = e.changedTouches[0].screenX; handleSwipeGesture(); }
function handleSwipeGesture() { const swipeThreshold = 50; if (touchEndX < touchStartX - swipeThreshold) { lightboxNext(); } else if (touchEndX > touchStartX + swipeThreshold) { lightboxPrev(); } }

/* ✨ MODIFIED SCRIPT: Removed the old click listener, as it's now handled by onclick in the HTML. */
document.addEventListener('keydown', function(e) { const lightbox = document.getElementById("lightbox-overlay"); if (lightbox.style.display === "block") { if (e.key === 'Escape') closeLightbox(); else if (e.key === 'ArrowLeft') lightboxPrev(); else if (e.key === 'ArrowRight') lightboxNext(); } });

window.addEventListener('resize', buildRotator);
document.addEventListener('DOMContentLoaded', buildRotator);
</script>
