<script>
// --- JavaScript Functions for Dynamic Scrolling ---
function calculateMarqueeDuration(titleElement) {
const textWidth = titleElement.offsetWidth;
const containerWidth = document.querySelector('#musictitle-container').offsetWidth;
const totalDistance = textWidth + containerWidth;
const scrollSpeed = 40;
const duration = totalDistance / scrollSpeed;
return Math.max(duration, 5);
}

function resetMarquee(titleEl) {
titleEl.style.animation = 'none';
titleEl.offsetWidth;
titleEl.style.animation = `scrollText ${calculateMarqueeDuration(titleEl)}s linear infinite`;
titleEl.style.animationPlayState = 'running';
}
// --- End Dynamic Scrolling Functions ---

let playlist = [
{ title: "Skeleton Punchers - This is a very long song title to properly test the scrolling functionality.", url: "https://dl.dropboxusercontent.com/scl/fi/r00ecru60czwms75519dr/Skeleton-Punchers.wav?rlkey=506p7uxpq9fcxojfddgixgjq0&st=l1e00654" }
];
let currentSongIndex = 0;
let skimInterval = null;
let isRandom = true;

const audio = document.querySelector('#musicsrc');
const playBtn = document.querySelector('#musicplayer .playpause');
const titleEl = document.querySelector('#currentSongTitle');
const titleContainer = document.querySelector('#musictitle');
const tooltipElement = document.getElementById('dynamicTooltip');
let tooltipTimeout;

function showTooltip(button, text) {
clearTimeout(tooltipTimeout);

const rect = button.getBoundingClientRect();
tooltipElement.innerText = text;

tooltipElement.style.left = `${rect.left + rect.width / 2}px`;
tooltipElement.style.top = `${rect.bottom + 5}px`;

tooltipElement.style.transform = `translateX(-50%)`;

tooltipElement.style.display = 'block';
tooltipElement.style.opacity = '1';
}

function hideTooltip() {
tooltipElement.style.opacity = '0';

tooltipTimeout = setTimeout(() => {
if (tooltipElement.style.opacity === '0') {
tooltipElement.style.display = 'none';
}
}, 300);
}

window.onload = async function() {
const nextBtn = document.querySelector('#musicplayer .next');
const prevBtn = document.querySelector('#musicplayer .prev');
const fastforwardBtn = document.querySelector('#musicplayer .fastforward');
const rewindBtn = document.querySelector('#musicplayer .rewind');
const randomBtn = document.querySelector('#musicplayer .random');
const speakerBtn = document.querySelector('#musicplayer .speaker');

if (!audio || !playBtn || !nextBtn || !prevBtn || !fastforwardBtn || !rewindBtn || !randomBtn || !speakerBtn || !titleEl || !titleContainer) {
titleEl.innerText = 'Player initialization failed';
return;
}

try {
const response = await fetch('https://raw.githubusercontent.com/CruxRiajuu/carrd-assets/main/playlist.json');
if (response.ok) {
playlist = await response.json();
currentSongIndex = Math.floor(Math.random() * playlist.length);
}
} catch (e) {
console.error('Failed to fetch playlist:', e);
}

audio.src = playlist[currentSongIndex].url;
titleEl.innerText = playlist[currentSongIndex].title;
audio.volume = 1.0;

document.querySelectorAll('#musiccontrols button').forEach(button => {
button.addEventListener('mouseenter', () => {
showTooltip(button, button.dataset.tooltip);
});
button.addEventListener('mouseleave', hideTooltip);
});

playBtn.addEventListener('click', () => playpause());
nextBtn.addEventListener('click', () => playNext());
prevBtn.addEventListener('click', () => playPrev());
fastforwardBtn.addEventListener('pointerdown', startFastForward);
fastforwardBtn.addEventListener('pointerup', stopSkim);
fastforwardBtn.addEventListener('pointerleave', stopSkim);
rewindBtn.addEventListener('pointerdown', startRewind);
rewindBtn.addEventListener('pointerup', stopSkim);
rewindBtn.addEventListener('pointerleave', stopSkim);
randomBtn.addEventListener('click', toggleRandom);
speakerBtn.addEventListener('click', () => toggleMute());

audio.addEventListener('ended', () => playNext());

/* MARQUEE START/STOP */
audio.addEventListener('pause', () => {
playBtn.innerHTML = '<i class="fas fa-play"></i>';
titleEl.style.animationPlayState = 'paused';
});
audio.addEventListener('play', () => {
titleContainer.style.display = 'block';
playBtn.innerHTML = '<i class="fas fa-pause"></i>';
titleEl.style.animation = `scrollText ${calculateMarqueeDuration(titleEl)}s linear infinite`;
titleEl.style.animationPlayState = 'running';
});
};

function playpause() {
if (audio.paused || audio.ended) {
audio.play().catch(e => {
playBtn.innerHTML = '<i class="fas fa-play"></i>';
});
titleContainer.style.display = 'block';

if (titleEl.style.animation === 'none' || titleEl.style.animation === '') {
titleEl.style.animation = `scrollText ${calculateMarqueeDuration(titleEl)}s linear infinite`;
}
titleEl.style.animationPlayState = 'running';
titleEl.innerText = playlist[currentSongIndex].title;
} else {
audio.pause();
}
}

function playNext() {
if (playlist.length === 0) return;
if (isRandom) {
currentSongIndex = Math.floor(Math.random() * playlist.length);
} else {
currentSongIndex = (currentSongIndex + 1) % playlist.length;
}
audio.src = playlist[currentSongIndex].url;
titleEl.innerText = playlist[currentSongIndex].title;

resetMarquee(titleEl);

audio.play().catch(e => {
titleEl.innerText = 'Error: ' + playlist[currentSongIndex].title;
});
titleContainer.style.display = 'block';
}

function playPrev() {
if (playlist.length === 0) return;
if (isRandom) {
currentSongIndex = Math.floor(Math.random() * playlist.length);
} else {
currentSongIndex = (currentSongIndex - 1 + playlist.length) % playlist.length;
}
audio.src = playlist[currentSongIndex].url;
titleEl.innerText = playlist[currentSongIndex].title;

resetMarquee(titleEl);

audio.play().catch(e => {
titleEl.innerText = 'Error: ' + playlist[currentSongIndex].title;
});
titleContainer.style.display = 'block';
}

function startFastForward() {
if (!audio || audio.paused) return;
stopSkim();
skimInterval = setInterval(() => {
audio.currentTime = Math.min(audio.currentTime + 0.5, audio.duration);
}, 100);
}

function startRewind() {
if (!audio || audio.paused) return;
stopSkim();
skimInterval = setInterval(() => {
audio.currentTime = Math.max(audio.currentTime - 0.5, 0);
}, 100);
}

function stopSkim() {
if (skimInterval) {
clearInterval(skimInterval);
skimInterval = null;
}
}

function toggleRandom() {
const randomBtn = document.querySelector('#musicplayer .random');
if (!randomBtn) return;
isRandom = !isRandom;
randomBtn.innerHTML = isRandom ? '<i class="fas fa-random"></i>' : '<i class="fas fa-list"></i>';
}

function toggleMute() {
const speakerIcon = document.querySelector('#musicplayer .speaker i');
if (!audio || !speakerIcon) return;
audio.muted = !audio.muted;
speakerIcon.className = audio.muted ? 'fas fa-volume-mute' : 'fas fa-volume-up';
}
