<style>
@import url('https://fonts.googleapis.com/css2?family=Oswald:wght@300;400&display=swap');
#reduce-motion-container { position: fixed; left: 0; right: 0; transition: transform 0.4s cubic-bezier(0.2, 0, 0, 1); transform: translateY(0); will-change: transform; background: #439CB6; width: 100%; padding: 5px 0; display: flex; justify-content: center; align-items: center; gap: 8px; }
#reduce-motion-container.hidden { transform: translateY(-565%); }
#reduce-motion-toggle { width: 36px; height: 18px; background: transparent; border: 1.5px solid #FFFFFF; border-radius: 9px; position: relative; cursor: pointer; transition: all 0.2s; }
#reduce-motion-toggle::before { content: ''; width: 12px; height: 12px; background: #FFFFFF; border-radius: 50%; position: absolute; left: 1.5px; top: 1.5px; transition: left 0.2s; }
#reduce-motion-toggle.active::before { left: 20.5px; }
#reduce-motion-label { color: #FFFFFF; font-size: 16px; font-family: 'Oswald', sans-serif; font-weight: 300; letter-spacing: 0.2px; text-transform: uppercase; cursor: pointer; }
.no-motion * { transition: none !important; animation: none !important; }
</style>
<div id="reduce-motion-container" data-fixed-element>
<label id="reduce-motion-label" for="reduce-motion-toggle">Reduce Motion</label>
<button id="reduce-motion-toggle" aria-labelledby="reduce-motion-label"></button>
</div>
<script>
document.addEventListener('DOMContentLoaded', () => {
const container = document.getElementById('reduce-motion-container');
const toggle = document.getElementById('reduce-motion-toggle');
const label = document.getElementById('reduce-motion-label');
const body = document.body;
let isInteracting = false;
let ticking = false;
let lastScrollY = window.scrollY;

if (container && toggle && body && label) {
const updateVisibility = () => {
const currentScrollY = window.scrollY;
if (!isInteracting) {
if (currentScrollY > lastScrollY && currentScrollY > 150) {
if (!container.classList.contains('hidden')) {
container.classList.add('hidden');
window.dispatchEvent(new Event('fixed_elements_update'));
}
} else if (currentScrollY < lastScrollY || currentScrollY <= 50) {
if (container.classList.contains('hidden')) {
container.classList.remove('hidden');
window.dispatchEvent(new Event('fixed_elements_update'));
}
}
}
lastScrollY = currentScrollY;
ticking = false;
};
const onScroll = () => { if (!ticking) { window.requestAnimationFrame(updateVisibility); ticking = true; } };
const handleInteractionStart = () => { isInteracting = true; container.classList.remove('hidden'); };
const handleInteractionEnd = () => { isInteracting = false; };

window.addEventListener('scroll', onScroll, { passive: true });
container.addEventListener('mouseenter', handleInteractionStart);
container.addEventListener('touchstart', handleInteractionStart, { passive: true });
container.addEventListener('mouseleave', handleInteractionEnd);
container.addEventListener('touchend', handleInteractionEnd);

const toggleMotion = () => {
const isActive = toggle.classList.toggle('active');
body.classList.toggle('no-motion', isActive);
localStorage.setItem('reduceMotion', isActive ? 'on' : 'off');
};
toggle.addEventListener('click', toggleMotion);
label.addEventListener('click', () => toggle.click());

const savedState = localStorage.getItem('reduceMotion');
if (savedState === 'on') {
body.classList.add('no-motion');
toggle.classList.add('active');
}
}
});
</script>
