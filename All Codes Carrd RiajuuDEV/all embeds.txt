// --- Marker Icon Positioning Logic ---

function updateMarkerPositions() {

starLinks.forEach(link => {

let textContent = '';

link.childNodes.forEach(node => {

if (node.nodeType === Node.TEXT_NODE && node.textContent.trim().length > 0) {

textContent += node.textContent.trim() + ' ';

}

});

textContent = textContent.trim();

if (textContent.length > 0) {

const span = document.createElement('span');

span.style.visibility = 'hidden';

span.style.position = 'absolute';

span.style.fontSize = getComputedStyle(link).fontSize;

span.style.fontFamily = getComputedStyle(link).fontFamily;

span.textContent = textContent;

document.body.appendChild(span);

const textWidth = span.offsetWidth;

document.body.removeChild(span);

link.style.setProperty('--text-width', `${textWidth}px`);

}

});

}

// --- Initial Setup and Events ---

function sendMessage(target) {

window.parent.postMessage({ starTarget: target }, '*');

}

starLinks.forEach(link => {

link.addEventListener('click', function() {

starLinks.forEach(otherLink => otherLink.classList.remove('nova-active'));

this.classList.add('nova-active');

const target = this.getAttribute('data-target');

sendMessage(target);

});

});

if (firstLink) {

firstLink.classList.add('nova-active');

const initialTarget = firstLink.getAttribute('data-target');

sendMessage(initialTarget);

}

updateMarkerPositions();

window.addEventListener('resize', updateMarkerPositions);

});

function playpause() {
if (audio.paused || audio.ended) {
audio.play().catch(e => {
playBtn.innerHTML = '<i class="fas fa-play"></i>';
});
titleContainer.style.display = 'block';

function playNext() {
if (playlist.length === 0) return;
if (isRandom) {
currentSongIndex = Math.floor(Math.random() * playlist.length);
} else {
currentSongIndex = (currentSongIndex + 1) % playlist.length;
}
audio.src = playlist[currentSongIndex].url;
titleEl.innerText = playlist[currentSongIndex].title;

resetMarquee(titleEl);

audio.play().catch(e => {
titleEl.innerText = 'Error: ' + playlist[currentSongIndex].title;
});
titleContainer.style.display = 'block';
}

function playPrev() {
if (playlist.length === 0) return;

currentSongIndex = Math.floor(Math.random() * playlist.length);

currentSongIndex = (currentSongIndex - 1 + playlist.length) % playlist.length;

titleEl.innerText = playlist[currentSongIndex].title;

resetMarquee(titleEl);

audio.play().catch(e => {
titleEl.innerText = 'Error: ' + playlist[currentSongIndex].title;
});
titleContainer.style.display = 'block';
}

}

}

}

card.style.opacity = '0.5'; card.style.zIndex = '5'; card.style.filter = 'brightness(0.5)'; card.classList.remove('active');
} else if (offset === 2) {
card.style.transform = 'translate(-50%, -50%) translateX(450px) translateZ(-400px) rotateY(-30deg) scale(0.6)';
card.style.opacity = '0.2'; card.style.zIndex = '1'; card.classList.remove('active');
} else {
card.style.transform = 'translate(-50%, -50%) translateX(1000px) scale(0)'; card.style.opacity = '0';
}
});
};

window.cyberConfirm = (title, msg, type) => {
return new Promise((resolve) => {
const modal = document.getElementById('cyber-confirm-overlay');
const box = document.getElementById('cyber-confirm-box');
const txtTitle = document.getElementById('cc-title');
const txtMsg = document.getElementById('cc-msg');
const btnYes = document.getElementById('cc-btn-yes');
const btnNo = document.getElementById('cc-btn-no');

box.classList.remove('confirm-green', 'confirm-red');
if(type === 'approve') { box.classList.add('confirm-green'); btnYes.innerText = "AUTHORIZE"; }
else { box.classList.add('confirm-red'); btnYes.innerText = "REJECT"; }

txtTitle.innerText = title; txtMsg.innerText = msg; modal.style.display = 'flex';

const handleYes = () => { cleanup(); resolve(true); };
const handleNo = () => { cleanup(); resolve(false); };
function cleanup() { modal.style.display = 'none'; btnYes.removeEventListener('click', handleYes); btnNo.removeEventListener('click', handleNo); }
btnYes.addEventListener('click', handleYes); btnNo.addEventListener('click', handleNo);
});
};

window.judgeGodUser = async (index, verdict) => {
const item = window.pendingList[index];
if(!item) return;

const title = verdict === 'approve' ? "AUTHORIZE USER?" : "REJECT USER?";
const msg = verdict === 'approve' ? `Grant ${item.type} status?` : `Reject application? Adds a strike.`;
const confirmed = await window.cyberConfirm(title, msg, verdict);
if(!confirmed) return;

const cardEl = document.getElementById(`card-${index}`);
if(verdict === 'approve') cardEl.classList.add('phase-approve');
else cardEl.classList.add('phase-reject');

window.rotatorIndex++;
updateGodCarousel();

const remaining = window.pendingList.length - window.rotatorIndex;
if(remaining <= 0) setTimeout(() => document.getElementById('god-empty-msg').style.display = 'block', 600);

try {
if(item.type === 'ID') {
if(verdict === 'approve') {
await window.db.collection('users').doc(item.id).update({ verification_status: 'verified', verified: true });
window.showToast(`${item.data.handle} APPROVED`, "success");
} else {
await window.db.collection('users').doc(item.id).update({
verification_status: 'rejected', verified: false,
rejection_count: firebase.firestore.FieldValue.increment(1)
});
window.showToast(`${item.data.handle} REJECTED`, "error");
}
}
else if(item.type === 'ARTIST') {
if(verdict === 'approve') {
await window.db.collection('users').doc(item.id).update({ artist_status: 'verified' });
window.showToast("Artist Approved", "success");
} else {
await window.db.collection('users').doc(item.id).update({ artist_status: 'rejected' });
window.showToast("Artist Rejected", "error");
}
}
} catch (e) {
window.showToast("DB ERROR: " + e.message, "error");
}
};
