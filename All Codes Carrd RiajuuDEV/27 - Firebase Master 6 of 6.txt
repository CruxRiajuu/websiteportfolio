<script>
// --- GLOBAL ADMIN VARIABLES ---
window.pendingList = [];
window.rotatorIndex = 0;

window.openAdminPanel = async () => {
if(!window.currentUserDoc.isAdmin) return window.showToast("ACCESS DENIED", "error");
document.getElementById('admin-ui').style.display = 'flex';
document.getElementById('user-dropdown').style.display = 'none';
await loadGodData();
};

window.closeAdminPanel = () => {
document.getElementById('admin-ui').style.display = 'none';
};

// --- 1. LOAD DATA (SCALABLE / PRODUCTION MODE) ---
window.loadGodData = async () => {
const container = document.getElementById('god-rotator');
const emptyMsg = document.getElementById('god-empty-msg');

container.innerHTML = '';
window.pendingList = [];
window.rotatorIndex = 0;

emptyMsg.style.display = 'block';
emptyMsg.innerText = "ACCESSING MAINFRAME...";

try {
// QUERY 1: Get Pending IDs (Only downloads pending docs)
const idSnaps = await window.db.collection('users')
.where('verification_status', '==', 'pending')
.limit(50) // Safety limit for batches
.get();

// QUERY 2: Get Pending Artists (Only downloads pending docs)
const artistSnaps = await window.db.collection('users')
.where('artist_status', '==', 'pending')
.limit(50)
.get();

idSnaps.forEach(doc => window.pendingList.push({ id: doc.id, data: doc.data(), type: 'ID' }));
artistSnaps.forEach(doc => window.pendingList.push({ id: doc.id, data: doc.data(), type: 'ARTIST' }));

document.getElementById('god-count').innerText = window.pendingList.length;

if(window.pendingList.length === 0) {
emptyMsg.innerText = "SYSTEM CLEAR";
emptyMsg.style.display = 'block';
} else {
emptyMsg.style.display = 'none';
buildGodCards();
updateGodCarousel();
}

} catch (e) {
console.error("FIREBASE INDEX ERROR:", e);

// --- AUTO-DIAGNOSIS FOR YOU ---
if(e.message.includes("index")) {
emptyMsg.innerHTML = `
<span style="color:red">MISSING INDEX</span><br>
<span style="font-size:0.5em; color:#aaa;">
1. Press <strong>F12</strong> (or Right Click -> Inspect)<br>
2. Go to "Console" Tab<br>
3. Click the long link generated by Firebase.<br>
4. It will build the index for you.
</span>`;
} else {
emptyMsg.innerText = "SYSTEM ERROR";
window.showToast("Error: " + e.message, "error");
}
}
};

// --- 2. BUILD CARDS ---
window.buildGodCards = () => {
const container = document.getElementById('god-rotator');
container.innerHTML = '';
const total = window.pendingList.length;

window.pendingList.forEach((item, index) => {
const card = document.createElement('div');
card.className = 'god-card';
card.id = `card-${index}`;

const color = item.type === 'ID' ? '#40c4ff' : '#b28dff';
card.style.borderTopColor = color;

const proofUrl = (item.type === 'ID') ? item.data.id_proof_url : item.data.artist_proof_url;
let mediaHtml = `<div style="color:#444;">NO MEDIA FOUND</div>`;

if(proofUrl) {
if(item.type === 'ID') {
mediaHtml = `<a href="${proofUrl}" target="_blank"><img src="${proofUrl}"></a>`;
} else {
if(proofUrl.includes('firebasestorage')) {
mediaHtml = `
<div style="display:flex; flex-direction:column; width:100%; height:100%;">
<video src="${proofUrl}" controls preload="metadata" style="flex:1; width:100%; object-fit:contain; background:#000;"></video>
<a href="${proofUrl}" target="_blank" style="color:#666; font-size:0.7rem; text-align:center; padding:5px; text-decoration:underline; font-family:monospace;">[OPEN RAW FILE]</a>
</div>`;
} else {
mediaHtml = `
<div style="display:flex; flex-direction:column; align-items:center; justify-content:center; height:100%; width:100%;">
<span style="color:#aaa; font-size:0.8rem; margin-bottom:10px; font-family:monospace;">EXTERNAL LINK</span>
<a href="${proofUrl}" target="_blank" class="cc-btn cc-yes" style="text-decoration:none; padding:10px 20px; flex:none; background:${color}; color:#000;">OPEN LINK</a>
</div>
`;
}
}
}

let joinDate = "Unknown";
if(item.data.created && item.data.created.toDate) joinDate = item.data.created.toDate().toLocaleDateString();

card.innerHTML = `
<div class="god-card-header" style="border-bottom:1px solid ${color};">
<h3 style="color:${color}">${item.data.handle}</h3>
<span>UID: ${item.id}</span>
</div>
<div class="god-card-body">
<div class="god-media-box">${mediaHtml}</div>
<div class="god-data-row"><span class="god-label">TYPE</span><span class="god-val" style="color:${color}">${item.type} VERIFICATION</span></div>
<div class="god-data-row"><span class="god-label">EMAIL</span><span class="god-val">${item.data.email}</span></div>
<div class="god-data-row"><span class="god-label">JOINED</span><span class="god-val">${joinDate}</span></div>
<div class="god-data-row"><span class="god-label">STRIKES</span><span class="god-val">${item.data.rejection_count || 0} / 5</span></div>
</div>
<div class="god-actions">
<button class="god-btn god-btn-reject" onclick="judgeGodUser(${index}, 'reject')">REJECT</button>
<button class="god-btn god-btn-approve" onclick="judgeGodUser(${index}, 'approve')">APPROVE</button>
</div>
<div class="god-card-footer" style="color:${color};">
QUEUE ITEM: ${index + 1} / ${total}
</div>
`;
container.appendChild(card);
});
};

// --- 3. CAROUSEL ENGINE ---
window.updateGodCarousel = () => {
const cards = document.querySelectorAll('.god-card');
cards.forEach((card, i) => {
let offset = i - window.rotatorIndex;
if (i < window.rotatorIndex) {
card.style.opacity = '0'; card.style.transform = 'translate(-50%, -50%) scale(0)'; card.style.zIndex = -1; return;
}
if (offset === 0) {
card.style.transform = 'translate(-50%, -50%) translateZ(0px) scale(1)';
card.style.opacity = '1'; card.style.zIndex = '10'; card.style.filter = 'none'; card.classList.add('active');
} else if (offset === 1) {
card.style.transform = 'translate(-50%, -50%) translateX(250px) translateZ(-200px) rotateY(-15deg) scale(0.8)';
card.style.opacity = '0.5'; card.style.zIndex = '5'; card.style.filter = 'brightness(0.5)'; card.classList.remove('active');
} else if (offset === 2) {
card.style.transform = 'translate(-50%, -50%) translateX(450px) translateZ(-400px) rotateY(-30deg) scale(0.6)';
card.style.opacity = '0.2'; card.style.zIndex = '1'; card.classList.remove('active');
} else {
card.style.transform = 'translate(-50%, -50%) translateX(1000px) scale(0)'; card.style.opacity = '0';
}
});
};

// --- 4. JUDGMENT LOGIC ---
window.cyberConfirm = (title, msg, type) => {
return new Promise((resolve) => {
const modal = document.getElementById('cyber-confirm-overlay');
const box = document.getElementById('cyber-confirm-box');
const txtTitle = document.getElementById('cc-title');
const txtMsg = document.getElementById('cc-msg');
const btnYes = document.getElementById('cc-btn-yes');
const btnNo = document.getElementById('cc-btn-no');

box.classList.remove('confirm-green', 'confirm-red');
if(type === 'approve') { box.classList.add('confirm-green'); btnYes.innerText = "AUTHORIZE"; }
else { box.classList.add('confirm-red'); btnYes.innerText = "REJECT"; }

txtTitle.innerText = title; txtMsg.innerText = msg; modal.style.display = 'flex';

const handleYes = () => { cleanup(); resolve(true); };
const handleNo = () => { cleanup(); resolve(false); };
function cleanup() { modal.style.display = 'none'; btnYes.removeEventListener('click', handleYes); btnNo.removeEventListener('click', handleNo); }
btnYes.addEventListener('click', handleYes); btnNo.addEventListener('click', handleNo);
});
};

window.judgeGodUser = async (index, verdict) => {
const item = window.pendingList[index];
if(!item) return;

const title = verdict === 'approve' ? "AUTHORIZE USER?" : "REJECT USER?";
const msg = verdict === 'approve' ? `Grant ${item.type} status?` : `Reject application? Adds a strike.`;
const confirmed = await window.cyberConfirm(title, msg, verdict);
if(!confirmed) return;

const cardEl = document.getElementById(`card-${index}`);
if(verdict === 'approve') cardEl.classList.add('phase-approve');
else cardEl.classList.add('phase-reject');

window.rotatorIndex++;
updateGodCarousel();

const remaining = window.pendingList.length - window.rotatorIndex;
if(remaining <= 0) setTimeout(() => document.getElementById('god-empty-msg').style.display = 'block', 600);

try {
if(item.type === 'ID') {
if(verdict === 'approve') {
await window.db.collection('users').doc(item.id).update({ verification_status: 'verified', verified: true });
window.showToast(`${item.data.handle} APPROVED`, "success");
} else {
await window.db.collection('users').doc(item.id).update({
verification_status: 'rejected', verified: false,
rejection_count: firebase.firestore.FieldValue.increment(1)
});
window.showToast(`${item.data.handle} REJECTED`, "error");
}
}
else if(item.type === 'ARTIST') {
if(verdict === 'approve') {
await window.db.collection('users').doc(item.id).update({ artist_status: 'verified' });
window.showToast("Artist Approved", "success");
} else {
await window.db.collection('users').doc(item.id).update({ artist_status: 'rejected' });
window.showToast("Artist Rejected", "error");
}
}
} catch (e) {
window.showToast("DB ERROR: " + e.message, "error");
}
};
</script>
