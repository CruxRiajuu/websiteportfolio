<script>
// 1. Hide on Scroll Logic
const topBarToHide = document.querySelector('.top-bar-nav');
if (topBarToHide) {
let lastScrollY = window.scrollY;
let ticking = false;
const updateVisibility = () => {
const currentScrollY = window.scrollY;
if (currentScrollY > lastScrollY && currentScrollY > 150) {
if (!topBarToHide.classList.contains('hidden')) {
topBarToHide.classList.add('hidden');
}
} else if (currentScrollY < lastScrollY || currentScrollY <= 50) {
if (topBarToHide.classList.contains('hidden')) {
topBarToHide.classList.remove('hidden');
}
}
lastScrollY = currentScrollY;
ticking = false;
};
const onScroll = () => {
if (!ticking) {
window.requestAnimationFrame(updateVisibility);
ticking = true;
}
};
window.addEventListener('scroll', onScroll, { passive: true });
}

// 2. Connect Cart Icon to Modal
document.addEventListener('DOMContentLoaded', function() {
const cartIcon = document.querySelector('.top-bar-nav .cart-icon');

const updateBadge = () => {
const savedCart = JSON.parse(localStorage.getItem('crux2025')||'[]');
const badge = document.getElementById('cart-item-count');
const count = savedCart.reduce((sum, item) => sum + item.qty, 0);

if (badge) {
if (count > 0) {
badge.textContent = count;
badge.classList.add('visible');
} else {
badge.classList.remove('visible');
}
}
};

updateBadge();

if (cartIcon) {
cartIcon.addEventListener('click', function() {
if (typeof openCartModal === 'function') {
openCartModal();
} else {
console.warn('Cart modal script not loaded yet.');
}
});
}

window.addEventListener('click', () => setTimeout(updateBadge, 500));
});
</script>
