<script src="https://www.youtube.com/iframe_api"></script>
<script>
let player1, player2;
function onPlayerStateChange(event) {
if (event.data === YT.PlayerState.PLAYING) {
const playingPlayerId = event.target.getIframe().id;
if (playingPlayerId === 'video-player-1' && player2) player2.pauseVideo();
else if (playingPlayerId === 'video-player-2' && player1) player1.pauseVideo();
}
}

function initLightboxController() {
const lightbox = document.getElementById('video-gallery-lightbox');
if (!lightbox) {
setTimeout(initLightboxController, 100);
return;
}

const closeBtn = lightbox.querySelector('.lightbox-close-btn');
const lightboxCarousel = lightbox.querySelector('.lightbox-carousel');

const openLightbox = (clickedVideoId, otherVideoId) => {
lightboxCarousel.innerHTML = `<div class="video-container"><div id="video-player-1"></div></div><div class="video-container"><div id="video-player-2"></div></div>`;
lightbox.classList.add('is-visible');
document.body.style.overflow = 'hidden';
setTimeout(() => {
initializePlayers(clickedVideoId, otherVideoId);
}, 50);
};

const initializePlayers = (firstVideoId, secondVideoId) => {
player1 = new YT.Player('video-player-1', { height: '100%', width: '100%', videoId: firstVideoId, playerVars: { 'autoplay': 1, 'rel': 0 }, events: { 'onStateChange': onPlayerStateChange } });
player2 = new YT.Player('video-player-2', { height: '100%', width: '100%', videoId: secondVideoId, playerVars: { 'rel': 0 }, events: { 'onStateChange': onPlayerStateChange } });
};

const closeLightbox = () => {
lightbox.classList.remove('is-visible');
if (player1) player1.destroy();
if (player2) player2.destroy();
player1 = null; player2 = null;
lightboxCarousel.innerHTML = '';
document.body.style.overflow = 'auto';
};

// This is the "receiver" that listens for the message from your iframe
window.addEventListener('message', function(event) {
// A basic security check for the origin can be added here if needed
// if (event.origin !== 'https://cruxriajuu.github.io') return;

if (event.data && event.data.type === 'open-video-lightbox') {
openLightbox(event.data.clickedId, event.data.otherId);
}
});

closeBtn.addEventListener('click', closeLightbox);
lightbox.addEventListener('click', (e) => {
if (e.target === lightbox) closeLightbox();
});
}

// Use the YouTube API's ready function to initialize our listener
window.onYouTubeIframeAPIReady = function() {
initLightboxController();
};
</script>
